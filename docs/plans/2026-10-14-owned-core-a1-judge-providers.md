**Goal:** Put every gate decision behind a provider choice (`jev` or a new deterministic `local` judge), log a shadow answer next to every Jev decision, and tag gate records with the orchestrated task, so the gate works with no key and Phase 4's baseline and shadow data start to accumulate.

A2 (task-level review set, scored starting tier, enforcement) builds on this data and is a separate plan.

```python
"""Task context for log records: the orchestrated task(s) active in a worktree, as "<branch>:<id>"
(the task id format /orchestrate uses for log_dispatch). Read-only; never raises; {} when nothing is active."""
import json
import os
from typing import Any, Dict, Optional


def _root(start: str) -> Optional[str]:
    d = os.path.abspath(start or ".")
    while True:
        if os.path.isfile(os.path.join(d, ".orchestrate", "active.json")):
            return d
        parent = os.path.dirname(d)
        if parent == d:
            return None
        d = parent


def branch(root: str) -> str:
    """The branch checked out in root, read from .git (a dir, or a worktree's "gitdir:" file). "" when unknown."""
    try:
        dot = os.path.join(root, ".git")
        if os.path.isfile(dot):
            with open(dot, encoding="utf-8") as f:
                first = f.readline().strip()
            if not first.startswith("gitdir:"):
                return ""
            gitdir = first[len("gitdir:"):].strip()
            dot = gitdir if os.path.isabs(gitdir) else os.path.join(root, gitdir)
        with open(os.path.join(dot, "HEAD"), encoding="utf-8") as f:
            head = f.readline().strip()
        prefix = "ref: refs/heads/"
        return head[len(prefix):] if head.startswith(prefix) else ""
    except Exception:
        return ""


def context(cwd: str) -> Dict[str, Any]:
    try:
        root = _root(cwd)
        if root is None:
            return {}
        with open(os.path.join(root, ".orchestrate", "active.json"), encoding="utf-8") as f:
            data = json.load(f)
        active = data.get("active", []) if isinstance(data, dict) else []
        ids = [str(a) for a in active if isinstance(a, (str, int))] if isinstance(active, list) else []
        if not ids:
            return {}
        b = branch(root) or "?"
        tasks = ["%s:%s" % (b, i) for i in ids]
        return {"task": tasks[0]} if len(tasks) == 1 else {"tasks": tasks[:16]}
    except Exception:
        return {}
```

```python
"""Local judge: deterministic answers to the gate's questions (safe, risk), no network, same answer shape as
jev_client.judge. A short DANGER list is judged unsafe (the gate denies); a short DESTRUCTIVE list is judged
destructive (the gate asks); everything else gets no decision and falls through to the harness's permissions.
It never allows: the gate itself never allows. An internal error answers destructive, so the gate asks."""
import re
import shlex
from typing import Any, Dict, List, Optional, Pattern, Tuple

_SEG = re.compile(r"&&|\|\||[;|\n]")
_ROOTS = ("", "~", "$HOME", "${HOME}", "/Users", "/home", "/*", "~/*", "C:", "c:")
_PREFIX = ("sudo", "command", "nohup", "env", "doas")

DANGER: List[Tuple[str, Pattern[str]]] = [
    ("fork_bomb", re.compile(r":\(\)\s*\{\s*:\s*\|\s*:\s*&\s*\}\s*;\s*:")),
    ("format_drive", re.compile(r"\bformat\s+[a-z]:", re.I)),
    ("del_drive", re.compile(r"\bdel\s+/s\s+/q\s+[a-z]:\\", re.I)),
    ("mkfs", re.compile(r"(?:^|[\s;&|(])mkfs(?:\.\w+)?\b")),
    ("dd_device", re.compile(r"\bdd\b[^;&|\n]*\bof=/dev/(?:sd|hd|nvme|disk|rdisk)")),
    ("overwrite_device", re.compile(r">\s*/dev/(?:sd|hd|nvme|disk|rdisk)\w*")),
    ("chmod_root", re.compile(r"\bchmod\s+-\w*R\w*\s+(?:0?777|a\+rwx)\s+/(?:\s|$)")),
]
DESTRUCTIVE: List[Tuple[str, Pattern[str]]] = [
    ("git_reset_hard", re.compile(r"\bgit\b[^;&|\n]*\breset\s+(?:\S+\s+)*--hard\b")),
    ("git_clean_force", re.compile(r"\bgit\b[^;&|\n]*\bclean\s+(?:\S+\s+)*-\w*f")),
    ("git_push_force", re.compile(r"\bgit\b[^;&|\n]*\bpush\b[^;&|\n]*(?:\s--force(?!-with-lease)\b|\s-\w*f\w*\b)")),
    ("git_branch_delete", re.compile(r"\bgit\b[^;&|\n]*\bbranch\s+(?:\S+\s+)*-D\b")),
    ("sql_drop", re.compile(r"\b(?:drop\s+(?:table|database|schema)|truncate\s+table)\b", re.I)),
    ("pipe_to_shell", re.compile(r"\b(?:curl|wget)\b[^;&\n]*\|\s*(?:sudo\s+)?(?:ba|z|da|k)?sh\b")),
]


def _rm_root(cmd: str) -> bool:
    """A recursive rm whose target is /, ~, $HOME or a home root (Ruflo pre-bash parity, plus home roots)."""
    for seg in _SEG.split(cmd):
        try:
            words = shlex.split(seg)
        except ValueError:
            words = seg.split()
        while words and ("=" in words[0] or words[0] in _PREFIX):
            words = words[1:]
        if not words or words[0].rsplit("/", 1)[-1] != "rm":
            continue
        args = words[1:]
        flags = "".join(w[1:] for w in args if w.startswith("-") and not w.startswith("--"))
        if not ("r" in flags.lower() or "--recursive" in args):
            continue
        if any(t.rstrip("/") in _ROOTS or t in ("/", "/*") for t in args if not t.startswith("-")):
            return True
    return False


def danger(cmd: str) -> Optional[str]:
    if _rm_root(cmd):
        return "rm_root"
    return next((name for name, rx in DANGER if rx.search(cmd)), None)


def destructive(cmd: str) -> Optional[str]:
    return next((name for name, rx in DESTRUCTIVE if rx.search(cmd)), None)


def _result(p_safe: float, risk: str, risk_conf: float, reasons: List[str]) -> Dict[str, Any]:
    return {"answers": {"safe": {"type": "noul", "value": p_safe, "confidence": max(p_safe, 1.0 - p_safe)},
                        "risk": {"type": "choice", "value": risk, "confidence": risk_conf, "probabilities": {}}},
            "reasons": reasons}


def gate_answers(tool: str, command: str = "", risky: bool = True) -> Dict[str, Any]:
    try:
        if tool == "Bash":
            hit = danger(command)
            if hit:
                return _result(0.02, "destructive", 0.95, ["danger:" + hit])
            hit = destructive(command)
            if hit:
                return _result(0.5, "destructive", 0.8, ["destructive:" + hit])
            if risky:
                return _result(0.8, "worth_a_look", 0.6, ["prescreen:risky"])
            return _result(0.9, "routine", 0.9, [])
        return _result(0.7, "worth_a_look", 0.6, ["file:sensitive_or_outside"])
    except Exception as e:
        return _result(0.5, "destructive", 0.8, ["error:" + type(e).__name__])
```

```python
"""Judge providers (owned core, sub-project A).

`provider` decides: "jev" (the remote model through jev_client) or "local" (local_judge, no network).
`shadow` answers too; its answer is logged next to the decision and never acted on. "auto": jev decides
when a key is set or a judge function is injected (tests), else local; the shadow is local when jev
decides, else none. Settings: thresholds `judge.provider`, `judge.shadow` (change with /tune apply)."""
import os
import sys
from typing import Any, Callable, Dict, Optional, Tuple

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))


def _has_key() -> bool:
    from jev_client import api_key  # lazy: the ssl/http import is paid only when a judge is needed
    return bool(api_key())


def resolve(th: Dict[str, Any], injected: bool) -> Tuple[str, str]:
    cfg = th.get("judge") if isinstance(th.get("judge"), dict) else {}
    provider, shadow = cfg.get("provider", "auto"), cfg.get("shadow", "auto")
    if provider not in ("local", "jev"):
        provider = "jev" if injected or _has_key() else "local"
    if shadow not in ("local", "jev", "none"):
        shadow = "local" if provider == "jev" else "none"
    return provider, ("none" if shadow == provider else shadow)


def ask(state: Any, questions: Dict[str, Any], local: Dict[str, Any], th: Dict[str, Any],
        judge_fn: Optional[Callable[..., Dict[str, Any]]] = None) -> Dict[str, Any]:
    provider, shadow = resolve(th, judge_fn is not None)
    out: Dict[str, Any] = {"provider": provider, "shadow_provider": shadow}
    remote: Any = None
    if "jev" in (provider, shadow):
        from jev_client import JevUnavailable, judge
        try:
            remote = (judge_fn or judge)(state, questions)
        except JevUnavailable as e:
            remote = e
    if provider == "jev" and not isinstance(remote, Exception):
        out.update(answers=remote["answers"], latency_ms=remote.get("latency_ms"),
                   input_tokens=remote.get("input_tokens"), cost=remote.get("cost"))
        if shadow == "local":
            out.update(shadow_answers=local["answers"], shadow_reasons=local.get("reasons", []))
        return out
    out.update(answers=local["answers"], reasons=local.get("reasons", []))
    if provider == "jev":  # Jev failed: the local judge decides this call
        out.update(provider="local", fallback="local", error=type(remote).__name__, shadow_provider="none")
    elif shadow == "jev":
        if isinstance(remote, Exception):
            out["shadow_error"] = type(remote).__name__
        else:
            out.update(shadow_answers=remote["answers"], shadow_cost=remote.get("cost"))
    return out
```

## Judge providers

The gate, the route hint and the `jev_judge` tool ask a judge. Two providers answer the same questions:

- **local** (no network, no key): deterministic rules in `claude/jev/local_judge.py`. Both default to `auto`: Jev decides when a key is set, with the local judge as its shadow; without a key the local judge decides. When Jev is unreachable the local judge decides that call (`fallback` in the log). `/tune` reports providers, fallbacks, shadow agreement and judged gate calls per task; change a setting with `/tune apply judge.provider '"local"'`.
