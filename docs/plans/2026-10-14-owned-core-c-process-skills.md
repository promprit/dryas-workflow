PROCESS_SKILLS = ("brainstorm", "worktree", "debug", "verify", "plan", "review-request", "review-receive", "tdd")
SKILL_FILES = ("UPSTREAM.md", "LICENSE-superpowers")  # provenance, copied as is
# Codex-specific instructions of the 1.x hand-written Codex skills and AGENTS.md: each must survive the rendering
HARNESS_REQUIRED: Dict[str, Tuple[str, ...]] = {
    "brainstorm": ("get the user's approval of a design",),
    "worktree": ("git worktree add ../<repo>-<topic> -b <branch>",),
    "debug": ("<dryas-root>/pstack/principles/fix-root-causes.md",),

_HARNESS = re.compile(r"^<!-- (/?)(claude|codex) -->\n?", re.M)
_FRONT = re.compile(r"\A---\n(.*?\n)---\n", re.S)
_CODEX_NAMES = (
    (re.compile(r"\$\{CLAUDE_PLUGIN_ROOT\}"), "<dryas-root>"),
    (re.compile(r"mcp__plugin_dryas_jev__"), ""),
    (re.compile(r"(?<![\w$/.@-])dryas:([a-z][a-z-]*)"), r"$dryas:\1"),
    (re.compile(r"(?<![\w$/.<-])/(orchestrate|wplan|wreview|commit|tune|interrogate)\b(?![/\w-]|\.\w)"), r"$dryas:\1"),
)


def harness_blocks(text: str, harness: str) -> str:
    """Keep the body of each <!-- HARNESS --> ... <!-- /HARNESS --> block for `harness`, drop the other's."""
    out, pos, open_name = [], 0, None
    for m in _HARNESS.finditer(text):
        closing, name = m.group(1) == "/", m.group(2)
        if closing != (open_name is not None) or (closing and name != open_name):
            raise ValueError("unbalanced <!-- %s%s --> marker" % (m.group(1), name))
        if not closing:
            out.append(text[pos:m.start()])
            open_name = name
        else:
            if open_name == harness:
                out.append(text[pos:m.start()])
            open_name = None
        pos = m.end()
    if open_name is not None:
        raise ValueError("unclosed <!-- %s --> marker" % open_name)
    out.append(text[pos:])
    return "".join(out)


def render_skill(text: str, harness: str) -> str:
    out = harness_blocks(text, harness)
    if harness != "codex":
        return out
    m = _FRONT.match(out)
    if m:  # Codex reads name and description only
        keep = [l for l in m.group(1).splitlines(True) if l.startswith(("name:", "description:"))]
        out = "---\n" + "".join(keep) + "---\n" + out[m.end():]
    for pat, rep in _CODEX_NAMES:
        out = pat.sub(rep, out)
    return out


def skill_plan(repo: Path) -> Dict[str, str]:
    """Target rel path under codex/skills -> its text, for every process skill and provenance file in claude/skills."""
    src, out = Path(repo) / "claude/skills", {}
    for name in SKILL_FILES:
        if (src / name).is_file():
            out["codex/skills/" + name] = (src / name).read_text(encoding="utf-8").replace("\r\n", "\n")
    for name in PROCESS_SKILLS:
        for f in sorted((src / name).glob("*.md")) if (src / name).is_dir() else []:
            try:
                text = render_skill(f.read_text(encoding="utf-8").replace("\r\n", "\n"), "codex")
            except ValueError as e:
                raise ValueError("%s: %s" % (f.relative_to(repo).as_posix(), e))
            out["codex/skills/%s/%s" % (name, f.name)] = text
    return out


def _skill_existing(repo: Path) -> List[str]:
    base, out = Path(repo) / "codex/skills", []
    out += ["codex/skills/" + n for n in SKILL_FILES if (base / n).is_file()]
    for name in PROCESS_SKILLS:
        if (base / name).is_dir():
            out += ["codex/skills/%s/%s" % (name, f.relative_to(base / name).as_posix())
                    for f in sorted((base / name).rglob("*")) if f.is_file()]
    return out


def harness_drift(repo: Path) -> List[str]:
    """One line per carried-over Codex instruction missing from a rendered process skill."""
    out = []
    for name, phrases in sorted(HARNESS_REQUIRED.items()):
        f = Path(repo) / "claude/skills" / name / "SKILL.md"
        if not f.is_file():
            continue
        text = render_skill(f.read_text(encoding="utf-8").replace("\r\n", "\n"), "codex")
        out += ['skill %s: the Codex copy lacks "%s"' % (name, p) for p in phrases if p not in text]
    return out

def drift(repo: Path) -> List[str]:
    want, out = plan(repo), []
    for rel, src in want.items():
        dest = Path(repo) / rel
        if not dest.is_file():
            out.append("missing " + rel)
        elif not _same(src, dest):
            out.append("differs " + rel)
    out += ["extra " + rel for rel in _existing(repo) if rel not in want]
    rendered = skill_plan(repo)
    for rel, text in rendered.items():
        dest = Path(repo) / rel
        if not dest.is_file():
            out.append("missing " + rel)
        elif dest.read_text(encoding="utf-8").replace("\r\n", "\n") != text:
            out.append("differs " + rel)
    out += ["extra " + rel for rel in _skill_existing(repo) if rel not in rendered]
    return sorted(out)


def sync(repo: Path) -> List[str]:
    """Make codex/ match claude/; return the changed rel paths."""
    want, rendered, changed = plan(repo), skill_plan(repo), []
    for line in drift(repo):
        kind, rel = line.split(" ", 1)
        dest = Path(repo) / rel
        if kind == "extra":
            dest.unlink()
        else:
            dest.parent.mkdir(parents=True, exist_ok=True)
            if rel in rendered:
                dest.write_bytes(rendered[rel].encode("utf-8"))
            else:
                shutil.copy2(str(want[rel]), str(dest))
        changed.append(rel)
    return changed

import os
import re
import sys
import unittest
from pathlib import Path
ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "tools"))
import sync_codex_plugin as scp

SKILLS = ROOT / "claude" / "skills"
UPSTREAM = SKILLS / "UPSTREAM.md"
UPSTREAM_OF = {"brainstorm": "brainstorming", "worktree": "using-git-worktrees", "debug": "systematic-debugging",
               "verify": "verification-before-completion", "plan": "writing-plans",
               "review-request": "requesting-code-review", "review-receive": "receiving-code-review",
               "tdd": "test-driven-development"}
PHRASE_FILES = {"template": ROOT / "claude/CLAUDE.md.template", "executor": ROOT / "claude/agents/executor.md",
                "orchestrate": SKILLS / "orchestrate/SKILL.md"}
SECTION = re.compile(r"^([a-z-]+/[A-Za-z0-9_.-]+\.md) § (.+)$")
PHRASE = re.compile(r'^(template|executor|orchestrate): "(.+)"$')
SP_NAME = re.compile(r"\bsuperpowers:[a-z]")
CLAUDE_ONLY = ("${CLAUDE_PLUGIN_ROOT}", "mcp__plugin_", "Skill(", "subagent_type", "EnterWorktree")
CLAUDE_COMMAND = re.compile(r"(?<![\w$/.<-])/(orchestrate|wplan|wreview|commit|tune|interrogate)\b(?![/\w-]|\.\w)")


def present():
    return [n for n in scp.PROCESS_SKILLS if (SKILLS / n / "SKILL.md").is_file()]


def tables():
    """Dryas skill -> list of (upstream cell, target cell) rows."""
    out, cur = {}, None
    for line in UPSTREAM.read_text(encoding="utf-8").splitlines():
        m = re.match(r"^## ([a-z-]+)$", line)
        if m:
            cur = out.setdefault(m.group(1), [])
        elif cur is not None and line.startswith("| ") and not line.startswith(("| Upstream item", "|---")):
            cells = [c.strip() for c in line.strip().strip("|").split("|")]
            cur.append(tuple(cells))
    return out


def upstream_dir():
    env = os.environ.get("DRYAS_SUPERPOWERS_SKILLS")
    if env:
        return Path(env)
    for p in sorted(Path.home().glob(".claude*/plugins/cache/claude-plugins-official/superpowers/6.4.1/skills")):
        return p
    return None


def headings(text):
    out, fence = [], None  # the open fence's backtick run: ```` blocks may hold ``` blocks
    for line in text.splitlines():
        m = re.match(r"^(`{3,})", line)
        if m:
            if fence is None:
                fence = m.group(1)
            elif line.strip() == fence:
                fence = None
        elif fence is None:
            m = re.match(r"^#{2,3} (.+?)\s*$", line)
            if m:
                out.append(m.group(1))
    return out


class ProcessSkillsTest(unittest.TestCase):
    def test_present_skills(self):
        # Task 2 lands brainstorm and plan; Tasks 3 and 4 extend this tuple until it is all eight
        self.assertEqual(set(present()), {"brainstorm", "plan"})

    def test_shape(self):
        for n in present():
            text = (SKILLS / n / "SKILL.md").read_text(encoding="utf-8")
            self.assertTrue(text.startswith("---\nname: %s\ndescription: " % n), n)
            desc = re.search(r"^description: (.+)$", text, re.M).group(1)
            self.assertLessEqual(len(desc), 1024, n)
            for f in (SKILLS / n).glob("*.md"):
                self.assertLess(len(f.read_text(encoding="utf-8").splitlines()), 500, f)

    def test_no_superpowers_names(self):
        files = [UPSTREAM] + [f for n in scp.PROCESS_SKILLS for f in (SKILLS / n).glob("*.md")]
        for f in files:
            for i, line in enumerate(f.read_text(encoding="utf-8").splitlines(), 1):
                self.assertIsNone(SP_NAME.search(line), (f.name, i, line))

    def test_codex_copies_have_no_claude_names(self):
        for rel, text in scp.skill_plan(ROOT).items():
            if rel.endswith(scp.SKILL_FILES):
                continue
            for s in CLAUDE_ONLY:
                self.assertNotIn(s, text, (rel, s))
            self.assertIsNone(CLAUDE_COMMAND.search(text), rel)

    def test_provenance(self):
        up = UPSTREAM.read_text(encoding="utf-8")
        self.assertIn("5bf4e78011075bcfc0dc295f0724994cd123ee71", up)
        self.assertIn("Jesse Vincent", (SKILLS / "LICENSE-superpowers").read_text(encoding="utf-8"))
        self.assertIn("claude/skills/UPSTREAM.md", (ROOT / "NOTICE.md").read_text(encoding="utf-8"))

    def test_every_present_skill_has_a_table(self):
        t = tables()
        for n in present():
            self.assertTrue(t.get(n), "no coverage table for %s in claude/skills/UPSTREAM.md" % n)

    def test_coverage_targets_resolve(self):
        for skill, rows in tables().items():
            for row in rows:
                self.assertEqual(len(row), 2, (skill, row))
                up, target = row
                self.assertTrue(up and target, (skill, row))
                for t in target.split("; "):
                    if t == "extra:superpowers":
                        continue
                    m, p = SECTION.match(t), PHRASE.match(t)
                    if m:
                        f = SKILLS / m.group(1)
                        self.assertTrue(f.is_file(), (skill, t))
                        self.assertIn(m.group(2), headings(f.read_text(encoding="utf-8"))
                                      + re.findall(r"^#{1,4} (.+?)\s*$", f.read_text(encoding="utf-8"), re.M), (skill, t))
                    elif p:
                        self.assertIn(p.group(2), PHRASE_FILES[p.group(1)].read_text(encoding="utf-8"), (skill, t))
                    else:
                        self.fail("bad target %r in %s" % (t, skill))

        if base is None:

        t = tables()
        for n in present():
            ups = [r[0] for r in t.get(n, [])]
            src = base / UPSTREAM_OF[n]
            for h in headings((src / "SKILL.md").read_text(encoding="utf-8")):
                self.assertTrue(any(u == h or u.startswith(h + " — ") for u in ups), "%s: no row for heading %r" % (n, h))
            for f in sorted(src.iterdir()):
                if f.name == "SKILL.md":
                    continue
                name = f.name + ("/" if f.is_dir() else "")
                self.assertTrue(any(u == name or u.startswith(name + " — ") for u in ups), "%s: no row for %s" % (n, name))
