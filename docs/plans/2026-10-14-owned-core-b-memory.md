# Owned core, plan B: memory and learning — Implementation Plan

> **For agentic workers:** run with `/orchestrate` on this file (Dryas). Each `## Task N:` section is self-contained: its Goal/Scope/Done/Failing-test lines are the `.orchestrate/PLAN.md` contract; Interfaces and Steps follow. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Give Dryas its own memory and learning layer: a stdlib SQLite store at `~/.dryas/dryas.db`, the hook stages that fill it (edits, tasks, sessions, Claude memory notes), session-end consolidation (dedupe, PageRank over the memory graph, explainable statistics), a read-only Ruflo import, and the `memory_search` / `memory_store` / `memory_list` / `memory_forget` / `swarm_init` tools on the existing `jev` MCP server. Every function the loop uses from Ruflo then has a Dryas-owned equivalent with a test, on Claude Code and on Codex.

Wiring the hooks into `chain.json`, the installer, the settings allow list and the skills is sub-project D.

**Tech Stack:** Python 3.9+ stdlib (`sqlite3`, FTS5 when the build has it), unittest classes run by pytest.

**Spec:** `docs/superpowers/specs/2026-10-14-owned-core-design.md` (Sub-project B: Store, Parity table, MCP tools, Migration; Decisions "One Python core, stdlib only", "Language-neutral contracts", "Additive only"; Error handling "Memory DB locked or missing", "Note import"; Testing "One parity test per row", "Migration").

- Files stay under 500 lines. Commits: conventional, explicit paths, never push. Shell probes that contain catastrophic command strings get denied by the live gate: keep such strings inside test files.

```python
"""Memory namespace: the folder name of the MAIN repo (worktrees map to it), else the basename of the start
dir, sanitized to [A-Za-z0-9._-] per character. Same rules and test vectors as claude/ruflo/ns.py
(claude/ruflo/tests/ns_cases.json), so Dryas memory and Ruflo memory agree on namespaces. A separate module
name because claude/jev and claude/ruflo are both on sys.path in one pytest run."""
import os
import re
import subprocess
from typing import List, Optional

_BAD = re.compile(r"[^A-Za-z0-9._-]")


def _parts(p: str) -> List[str]:
    return p.replace("\\", "/").rstrip("/").split("/")


def sanitize(name: str) -> str:
    name = _BAD.sub("_", name)
    return "_unknown" if name in ("", ".", "..") else name


def ns_from(start: str, common_dir: Optional[str]) -> str:
    name = None
    if common_dir:
        c = _parts(common_dir)
        if len(c) >= 2 and c[-1] == ".git" and c[-2]:
            name = c[-2]
    if name is None:
        name = _parts(start)[-1]
    return sanitize(name)


def common_dir(start: str) -> Optional[str]:
    try:
        p = subprocess.run(["git", "-C", start, "rev-parse", "--path-format=absolute", "--git-common-dir"],
                           capture_output=True, text=True, timeout=5)
    except (OSError, subprocess.SubprocessError):
        return None
    out = p.stdout.strip()
    return out if p.returncode == 0 and out else None


def main_root(start: str) -> Optional[str]:
    """The main repo's root (a worktree maps to it); None outside git or for a bare repo."""
    c = common_dir(start)
    if c and _parts(c)[-1] == ".git":
        return os.path.dirname(c.replace("\\", "/").rstrip("/"))
    return None


def project_ns(start: Optional[str] = None) -> str:
    start = start or os.environ.get("CLAUDE_PROJECT_DIR") or os.getcwd()
    return ns_from(start, common_dir(start))
```

`claude/jev/memdb.py`:

```python
"""Dryas memory store: one SQLite file in WAL mode, stdlib sqlite3 (owned core, sub-project B).

The schema is a language-neutral contract; other tools may read it. migrate() is idempotent and keyed by
PRAGMA user_version; a database written by a newer Dryas opens read-only. MEM-5: when this sqlite build has
no FTS5 (or DRYAS_NO_FTS=1) search falls back to LIKE and the FTS triggers are dropped so writes keep
working; the index is rebuilt the next time a connection has FTS5."""
import contextlib
import hashlib
import os
import sqlite3
import sys
import urllib.request
from typing import Iterator, Optional

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import paths

SCHEMA_VERSION = 1
BUSY_MS = 1500

TABLES = (
    """CREATE TABLE IF NOT EXISTS memory (
        id INTEGER PRIMARY KEY, ns TEXT NOT NULL, kind TEXT NOT NULL, title TEXT NOT NULL DEFAULT '',
        body TEXT NOT NULL DEFAULT '', tags TEXT NOT NULL DEFAULT '[]', task_id TEXT, "commit" TEXT,
        source TEXT NOT NULL DEFAULT 'dryas', key TEXT, hash TEXT NOT NULL, embedding BLOB,
        meta TEXT NOT NULL DEFAULT '{}', created REAL NOT NULL, accessed REAL, hits INTEGER NOT NULL DEFAULT 0)""",
    "CREATE UNIQUE INDEX IF NOT EXISTS memory_key ON memory(ns, source, key) WHERE key IS NOT NULL",
    "CREATE INDEX IF NOT EXISTS memory_ns_kind ON memory(ns, kind, created)",
    "CREATE INDEX IF NOT EXISTS memory_hash ON memory(ns, hash)",
    """CREATE TABLE IF NOT EXISTS edits (ns TEXT NOT NULL, session TEXT, task_id TEXT, file TEXT NOT NULL,
        ok INTEGER NOT NULL, ts REAL NOT NULL)""",
    "CREATE INDEX IF NOT EXISTS edits_ns ON edits(ns, ts)",
    """CREATE TABLE IF NOT EXISTS tasks (task_id TEXT PRIMARY KEY, ns TEXT NOT NULL, title TEXT NOT NULL DEFAULT '',
        scope TEXT NOT NULL DEFAULT '[]', tier_start TEXT, tier_end TEXT, outcome TEXT, started REAL, ended REAL,
        meta TEXT NOT NULL DEFAULT '{}')""",
    "CREATE TABLE IF NOT EXISTS sessions (id TEXT PRIMARY KEY, ns TEXT NOT NULL, started REAL, ended REAL, summary TEXT)",
    """CREATE TABLE IF NOT EXISTS patterns (ns TEXT NOT NULL, kind TEXT NOT NULL, key TEXT NOT NULL, value TEXT,
        support INTEGER NOT NULL DEFAULT 0, updated REAL NOT NULL, PRIMARY KEY (ns, kind, key))""",
    "CREATE TABLE IF NOT EXISTS access (session TEXT NOT NULL, memory_id INTEGER NOT NULL, ts REAL NOT NULL)",
    "CREATE INDEX IF NOT EXISTS access_mem ON access(memory_id)",
    """CREATE TABLE IF NOT EXISTS forgotten (ns TEXT NOT NULL, source TEXT NOT NULL, key TEXT NOT NULL,
        ts REAL NOT NULL, PRIMARY KEY (ns, source, key))""",
    "CREATE TABLE IF NOT EXISTS kv (key TEXT PRIMARY KEY, value TEXT)",
)
FTS = ("CREATE VIRTUAL TABLE IF NOT EXISTS memory_fts USING fts5(title, body, tags, content='memory', "
       "content_rowid='id')")
TRIGGERS = (
    """CREATE TRIGGER IF NOT EXISTS memory_ai AFTER INSERT ON memory BEGIN
        INSERT INTO memory_fts(rowid, title, body, tags) VALUES (new.id, new.title, new.body, new.tags); END""",
    """CREATE TRIGGER IF NOT EXISTS memory_ad AFTER DELETE ON memory BEGIN
        INSERT INTO memory_fts(memory_fts, rowid, title, body, tags)
        VALUES ('delete', old.id, old.title, old.body, old.tags); END""",
    """CREATE TRIGGER IF NOT EXISTS memory_au AFTER UPDATE OF title, body, tags ON memory BEGIN
        INSERT INTO memory_fts(memory_fts, rowid, title, body, tags)
        VALUES ('delete', old.id, old.title, old.body, old.tags);
        INSERT INTO memory_fts(rowid, title, body, tags) VALUES (new.id, new.title, new.body, new.tags); END""",
)
TRIGGER_NAMES = ("memory_ai", "memory_ad", "memory_au")


class Conn(sqlite3.Connection):
    """A connection that knows whether FTS5 search is usable and whether it may write."""
    fts = False
    readonly = False


def db_path() -> str:
    return os.environ.get("DRYAS_DB") or os.path.join(paths.data_root(), "dryas.db")


def content_hash(title: str, body: str) -> str:
    norm = " ".join(("%s\n%s" % (title, body)).split()).lower()
    return hashlib.sha256(norm.encode("utf-8")).hexdigest()


def fts5_available(conn: sqlite3.Connection) -> bool:
    if os.environ.get("DRYAS_NO_FTS") == "1":
        return False
    try:
        conn.execute("CREATE VIRTUAL TABLE temp.fts5_probe USING fts5(x)")
        conn.execute("DROP TABLE temp.fts5_probe")
        return True
    except sqlite3.Error:
        return False


@contextlib.contextmanager
def tx(conn: sqlite3.Connection) -> Iterator[sqlite3.Connection]:
    """One write transaction. Takes the write lock up front, so a busy store fails at BEGIN, not halfway."""
    conn.execute("BEGIN IMMEDIATE")
    try:
        yield conn
    except BaseException:
        conn.execute("ROLLBACK")
        raise
    conn.execute("COMMIT")


def _kv(conn: sqlite3.Connection, key: str) -> Optional[str]:
    row = conn.execute("SELECT value FROM kv WHERE key = ?", (key,)).fetchone()
    return row[0] if row else None


def _has_table(conn: sqlite3.Connection, name: str) -> bool:
    return conn.execute("SELECT 1 FROM sqlite_master WHERE name = ?", (name,)).fetchone() is not None


def migrate(conn: Conn) -> None:
    with tx(conn):
        for stmt in TABLES:
            conn.execute(stmt)
        if conn.fts:
            conn.execute(FTS)
            for stmt in TRIGGERS:
                conn.execute(stmt)
            if _kv(conn, "fts") != "1":  # new store, or rows written while FTS5 was missing
                conn.execute("INSERT INTO memory_fts(memory_fts) VALUES ('rebuild')")
                conn.execute("INSERT OR REPLACE INTO kv(key, value) VALUES ('fts', '1')")
        else:
            for name in TRIGGER_NAMES:
                conn.execute("DROP TRIGGER IF EXISTS %s" % name)
            conn.execute("INSERT OR REPLACE INTO kv(key, value) VALUES ('fts', '0')")
        conn.execute("PRAGMA user_version = %d" % SCHEMA_VERSION)


def _open(target: str, busy_ms: int, uri: bool = False) -> Conn:
    conn = sqlite3.connect(target, timeout=busy_ms / 1000.0, isolation_level=None, factory=Conn, uri=uri)
    conn.row_factory = sqlite3.Row
    return conn


def connect(path: Optional[str] = None, busy_ms: int = BUSY_MS) -> Conn:
    """Open the store, creating or migrating it. Raises sqlite3.Error or OSError; callers degrade (MEM-5)."""
    p = path or db_path()
    fresh = not os.path.exists(p)
    os.makedirs(os.path.dirname(os.path.abspath(p)), mode=0o700, exist_ok=True)
    conn = _open(p, busy_ms)
    ver = conn.execute("PRAGMA user_version").fetchone()[0]
    if ver > SCHEMA_VERSION:  # written by a newer Dryas: read, never write
        conn.close()
        conn = _open("file:%s?mode=ro" % urllib.request.pathname2url(os.path.abspath(p)), busy_ms, uri=True)
        conn.readonly = True
        conn.fts = fts5_available(conn) and _has_table(conn, "memory_fts")
        return conn
    conn.fts = fts5_available(conn)
    conn.execute("PRAGMA journal_mode=WAL")
    if not (ver == SCHEMA_VERSION and _kv(conn, "fts") == ("1" if conn.fts else "0")):
        migrate(conn)
    if fresh and os.name == "posix" and os.path.exists(p):
        os.chmod(p, 0o600)  # portable-ok: POSIX file privacy; guarded by os.name
    return conn
```

```python
"""Memory rows (owned core, sub-project B): store with the MEM-4 checks, search with one explainable ranking
on FTS5 and on the LIKE fallback, list, forget.

Ranking: the share of query terms matched (a term matches when it starts one of the row's words), then the
PageRank of the memory graph and how often a memory was used, both written by memlearn.consolidate. Every
result carries a reason string; search also returns the learned patterns whose key mentions a query term."""
import json
import os
import re
import sys
import time
from typing import Any, Dict, Iterable, List, Optional, Tuple

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import memdb
import redact

MAX_TITLE = 200
MAX_BODY = 4000
STORE_KINDS = ("outcome", "fact")
RANK_WEIGHT = 0.3
HIT_WEIGHT = 0.1
CANDIDATES = 200
LIKE_CANDIDATES = 2000
MAX_TERMS = 16
_WORD = re.compile(r"[^\W_]+")
_DIFF = re.compile(r"^(?:diff --git |@@ -\d+(?:,\d+)? \+\d+|\+\+\+ [ab/]|--- a/|index [0-9a-f]{7,}\.\.[0-9a-f]{7,})", re.M)


class Rejected(ValueError):
    """MEM-4: this content may not be stored."""


def check(title: str, body: str) -> None:
    if not title.strip() and not body.strip():
        raise Rejected("nothing to store: title and body are empty")
    if len(title) > MAX_TITLE:
        raise Rejected("title is %d characters; keep it under %d" % (len(title), MAX_TITLE))
    if len(body) > MAX_BODY:
        raise Rejected("body is %d characters, the cap is %d: store what worked and what failed, not file contents"
                       % (len(body), MAX_BODY))
    if _DIFF.search(body):
        raise Rejected("body looks like a diff: store the outcome, not the patch")
    for text in (title, body):
        if redact.redact(text) != text:
            raise Rejected("contains a secret-shaped value (MEM-4); remove it and store again")


def store(conn, ns: str, title: str, body: str, kind: str = "outcome", tags: Iterable[str] = (),
          task_id: Optional[str] = None, commit: Optional[str] = None, source: str = "dryas",
          key: Optional[str] = None, created: Optional[float] = None, meta: Optional[Dict[str, Any]] = None,
          check_content: bool = True) -> Dict[str, Any]:
    title, body = (title or "").strip(), (body or "").strip()
    if check_content:
        check(title, body)
    h = memdb.content_hash(title, body)
    tag_list = [str(t)[:64] for t in (tags or ())][:20]
    with memdb.tx(conn):
        if key is None:
            same = conn.execute("SELECT id FROM memory WHERE ns = ? AND hash = ? AND kind != 'note' LIMIT 1",
                                (ns, h)).fetchone()
            if same:
                return {"id": same["id"], "stored": False, "duplicate": True}
        cur = conn.execute(
            'INSERT OR IGNORE INTO memory (ns, kind, title, body, tags, task_id, "commit", source, key, hash, meta, created) '
            "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
            (ns, kind, title, body, json.dumps(tag_list), task_id, commit, source, key, h, json.dumps(meta or {}),
             time.time() if created is None else created))
        if cur.rowcount == 0:
            row = conn.execute("SELECT id FROM memory WHERE ns = ? AND source = ? AND key = ?", (ns, source, key)).fetchone()
            return {"id": row["id"], "stored": False, "duplicate": True}
        return {"id": cur.lastrowid, "stored": True, "duplicate": False}


def terms(query: str) -> List[str]:
    out: List[str] = []
    for t in _WORD.findall((query or "").lower()):
        if len(t) >= 2 and t not in out:
            out.append(t)
    return out[:MAX_TERMS]


def _iso(ts: Optional[float]) -> Optional[str]:
    return None if ts is None else time.strftime("%Y-%m-%dT%H:%M:%S%z", time.localtime(ts))


def _tags(text: Optional[str]) -> List[Any]:
    try:
        v = json.loads(text or "[]")
        return v if isinstance(v, list) else []
    except ValueError:
        return []


def _result(r, score: Optional[float], reason: str, body_cap: Optional[int] = None) -> Dict[str, Any]:
    body = r["body"] if body_cap is None else r["body"][:body_cap]
    return {"id": r["id"], "kind": r["kind"], "title": r["title"], "body": body, "tags": _tags(r["tags"]),
            "task_id": r["task_id"], "commit": r["commit"], "source": r["source"], "created": _iso(r["created"]),
            "score": None if score is None else round(score, 3), "reason": reason}


def _candidates(conn, ns: str, ts: List[str], include_archive: bool) -> List[Any]:
    arch = "" if include_archive else " AND m.kind != 'archive'"
    if conn.fts:  # terms are letters and digits only, so quoting makes each a plain prefix query
        q = " OR ".join('"%s"*' % t for t in ts)
        return conn.execute("SELECT m.* FROM memory_fts JOIN memory m ON m.id = memory_fts.rowid "
                            "WHERE memory_fts MATCH ? AND m.ns = ?" + arch + " ORDER BY bm25(memory_fts) LIMIT ?",
                            (q, ns, CANDIDATES)).fetchall()
    like = " OR ".join("(m.title LIKE ? OR m.body LIKE ? OR m.tags LIKE ?)" for _ in ts)
    args: List[Any] = [ns]
    for t in ts:
        args += ["%" + t + "%"] * 3
    return conn.execute("SELECT m.* FROM memory m WHERE m.ns = ?" + arch + " AND (" + like + ") "
                        "ORDER BY m.created DESC LIMIT ?", args + [LIKE_CANDIDATES]).fetchall()


def _ranks(conn, ns: str, ids: List[int]) -> Tuple[Dict[int, float], float]:
    top = conn.execute("SELECT MAX(CAST(value AS REAL)) FROM patterns WHERE ns = ? AND kind = 'rank'", (ns,)).fetchone()[0]
    if not ids or not top:
        return {}, 0.0
    marks = ",".join("?" * len(ids))
    rows = conn.execute("SELECT key, value FROM patterns WHERE ns = ? AND kind = 'rank' AND key IN (%s)" % marks,
                        [ns] + [str(i) for i in ids]).fetchall()
    return {int(r[0]): float(r[1]) for r in rows}, float(top)


def _reason(matched: int, total: int, rank: float, hits: int) -> str:
    out = "matched %d of %d terms" % (matched, total)
    if rank > 0:
        out += "; graph rank %.2f" % rank
    if hits:
        out += "; used %d times" % hits
    return out


def pattern_reason(kind: str, key: str, value: Optional[str], support: int) -> str:
    if kind == "file_fail":
        return "%s: %.0f%% of %d edits failed" % (key, float(value or 0) * 100, support)
    if kind == "coedit":
        a, _, b = key.partition("|")
        return "%s and %s were edited together in %d tasks" % (a, b, support)
    if kind == "tier":
        return "%s tasks finished at tier %s (%d tasks)" % (key, value, support)
    return "%s %s: %s (%d)" % (kind, key, value, support)


def patterns_for(conn, ns: str, ts: List[str], limit: int = 5) -> List[Dict[str, Any]]:
    if not ts:
        return []
    out = []
    for r in conn.execute("SELECT kind, key, value, support FROM patterns WHERE ns = ? AND kind IN "
                          "('file_fail', 'coedit', 'tier') ORDER BY support DESC, kind, key LIMIT 500", (ns,)):
        if any(t in r["key"].lower() for t in ts):
            out.append({"kind": r["kind"], "key": r["key"], "value": r["value"], "support": r["support"],
                        "reason": pattern_reason(r["kind"], r["key"], r["value"], r["support"])})
            if len(out) >= limit:
                break
    return out


def _touch(conn, ids: List[int], session: Optional[str]) -> None:
    if not ids or conn.readonly:
        return
    now = time.time()
    try:
        with memdb.tx(conn):
            conn.execute("UPDATE memory SET hits = hits + 1, accessed = ? WHERE id IN (%s)" % ",".join("?" * len(ids)),
                         [now] + ids)
            if session:
                conn.executemany("INSERT INTO access (session, memory_id, ts) VALUES (?, ?, ?)",
                                 [(session, i, now) for i in ids])
    except Exception:
        pass  # a busy store must not cost the caller its results


def search(conn, ns: str, query: str, limit: int = 3, include_archive: bool = False,
           session: Optional[str] = None) -> Dict[str, Any]:
    limit = max(1, min(int(limit), 20))
    ts = terms(query)
    if not ts:
        arch = "" if include_archive else " AND kind != 'archive'"
        rows = conn.execute("SELECT * FROM memory WHERE ns = ?" + arch + " ORDER BY created DESC, id DESC LIMIT ?",
                            (ns, limit)).fetchall()
        hits = [(0.0, r, "most recent (no search terms)") for r in rows]
    else:
        rows = _candidates(conn, ns, ts, include_archive)
        ranks, top = _ranks(conn, ns, [r["id"] for r in rows])
        scored = []
        for r in rows:
            words = set(_WORD.findall(("%s %s %s" % (r["title"], r["body"], r["tags"])).lower()))
            matched = sum(1 for t in ts if any(w.startswith(t) for w in words))
            if not matched:
                continue
            rank = ranks.get(r["id"], 0.0) / top if top else 0.0
            used = r["hits"] or 0
            score = matched / len(ts) + RANK_WEIGHT * rank + HIT_WEIGHT * min(used, 10) / 10.0
            scored.append((score, r, _reason(matched, len(ts), rank, used)))
        scored.sort(key=lambda x: (-x[0], -(x[1]["created"] or 0), x[1]["id"]))
        hits = scored[:limit]
    _touch(conn, [r["id"] for _, r, _ in hits], session)
    return {"results": [_result(r, s, why) for s, r, why in hits], "patterns": patterns_for(conn, ns, ts),
            "fts": bool(conn.fts)}


def list_entries(conn, ns: str, kind: Optional[str] = None, limit: int = 20, offset: int = 0,
                 include_archive: bool = False) -> List[Dict[str, Any]]:
    sql, args = "SELECT * FROM memory WHERE ns = ?", [ns]
    if kind:
        sql += " AND kind = ?"
        args.append(kind)
    elif not include_archive:
        sql += " AND kind != 'archive'"
    sql += " ORDER BY created DESC, id DESC LIMIT ? OFFSET ?"
    args += [max(1, min(int(limit), 100)), max(0, int(offset))]
    return [_result(r, None, "listed", body_cap=300) for r in conn.execute(sql, args)]


def forget(conn, ns: str, mem_id: int) -> str:
```

```python
"""Session-end consolidation (owned core, sub-project B): dedupe, the memory graph, PageRank, and the
explainable statistics that memory_search returns and the local judge's history signal reads.

Graph nodes are memories ("m:<id>") and files ("f:<path>"). Edges: two memories recalled in the same session
(co-access), two files edited in the same task or, without a task, the same session (co-edit), and a memory
and the files its task edited. Pattern rows of kinds rank, file_fail, coedit and tier are replaced for the
namespace on every run."""
import os
import re
import sys
import time
from collections import Counter, defaultdict
from itertools import combinations
from typing import Any, Dict, List, Optional, Set, Tuple

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import memdb

MIN_EDITS = 3
MIN_SUPPORT = 2
MAX_GROUP = 50  # files or memories per group that form pairs; caps the n^2 growth of one huge task
ACCESS_DAYS = 180
EDIT_DAYS = 365
KINDS = ("rank", "file_fail", "coedit", "tier")
_TYPE = re.compile(r"\s*([A-Za-z]+)")


def pagerank(edges: Dict[Tuple[str, str], float], damping: float = 0.85, iters: int = 100,
             tol: float = 1e-10) -> Dict[str, float]:
    nbrs: Dict[str, Dict[str, float]] = defaultdict(dict)
    for (a, b), w in edges.items():
        if a == b or w <= 0:
            continue
        nbrs[a][b] = nbrs[a].get(b, 0.0) + w
        nbrs[b][a] = nbrs[b].get(a, 0.0) + w
    nodes = sorted(nbrs)
    n = len(nodes)
    if not n:
        return {}
    rank = {v: 1.0 / n for v in nodes}
    out_w = {v: sum(nbrs[v].values()) for v in nodes}
    for _ in range(iters):
        new = {v: (1.0 - damping) / n for v in nodes}
        for v in nodes:
            share = damping * rank[v] / out_w[v]
            for u, w in nbrs[v].items():
                new[u] += share * w
        delta = sum(abs(new[v] - rank[v]) for v in nodes)
        rank = new
        if delta < tol:
            break
    return rank


def task_type(title: str) -> str:
    m = _TYPE.match(title or "")
    return m.group(1).lower() if m else "other"


def _dedupe(conn, ns: str) -> int:
    where = "ns = ? AND kind NOT IN ('note', 'archive') AND source != 'ruflo'"
    removed = 0
    for (h,) in conn.execute("SELECT hash FROM memory WHERE " + where + " GROUP BY hash HAVING COUNT(*) > 1",
                             (ns,)).fetchall():
        rows = conn.execute("SELECT id, hits FROM memory WHERE " + where + " AND hash = ? ORDER BY created, id",
                            (ns, h)).fetchall()
        keep, rest = rows[0][0], [r[0] for r in rows[1:]]
        marks = ",".join("?" * len(rest))
        conn.execute("UPDATE memory SET hits = ? WHERE id = ?", (sum(r[1] or 0 for r in rows), keep))
        conn.execute("UPDATE access SET memory_id = ? WHERE memory_id IN (%s)" % marks, [keep] + rest)
        conn.execute("DELETE FROM memory WHERE id IN (%s)" % marks, rest)
        removed += len(rest)
    return removed


def _pairs(items: Set[str]) -> List[Tuple[str, str]]:
    return list(combinations(sorted(items)[:MAX_GROUP], 2))


def consolidate(conn, ns: str, now: Optional[float] = None) -> Dict[str, int]:
    now = time.time() if now is None else now
    with memdb.tx(conn):
        deduped = _dedupe(conn, ns)
        conn.execute("DELETE FROM access WHERE ts < ?", (now - ACCESS_DAYS * 86400,))
        edges: Counter = Counter()

        by_session: Dict[str, Set[str]] = defaultdict(set)
        for s, mid in conn.execute("SELECT a.session, a.memory_id FROM access a JOIN memory m ON m.id = a.memory_id "
                                   "WHERE m.ns = ?", (ns,)):
            by_session[s].add("m:%d" % mid)
        for ids in by_session.values():
            for pair in _pairs(ids):
                edges[pair] += 1

        groups: Dict[Tuple[str, str], Set[str]] = defaultdict(set)
        totals: Counter = Counter()
        fails: Counter = Counter()
        for s, t, f, ok in conn.execute("SELECT session, task_id, file, ok FROM edits WHERE ns = ? AND ts >= ?",
                                        (ns, now - EDIT_DAYS * 86400)):
            groups[("task", t) if t else ("session", s or "")].add(f)
            totals[f] += 1
            fails[f] += 0 if ok else 1
        coedit: Counter = Counter()
        for files in groups.values():
            for a, b in _pairs(files):
                coedit[(a, b)] += 1
                edges[("f:" + a, "f:" + b)] += 1
        for mid, tid in conn.execute("SELECT id, task_id FROM memory WHERE ns = ? AND task_id IS NOT NULL", (ns,)):
            for f in sorted(groups.get(("task", tid), ()))[:MAX_GROUP]:
                edges[tuple(sorted(("m:%d" % mid, "f:" + f)))] += 1

        ranks = pagerank(dict(edges))
        degree: Counter = Counter()
        for a, b in edges:
            degree[a] += 1
            degree[b] += 1
        tiers: Dict[str, Counter] = defaultdict(Counter)
        for title, tier in conn.execute("SELECT title, tier_end FROM tasks WHERE ns = ? AND tier_end IS NOT NULL "
                                        "AND ended IS NOT NULL", (ns,)):
            tiers[task_type(title)][tier] += 1

        rows: List[Tuple[Any, ...]] = []
        rows += [(ns, "rank", v[2:], "%.6f" % r, degree[v], now) for v, r in sorted(ranks.items()) if v.startswith("m:")]
        ranked = len(rows)
        rows += [(ns, "file_fail", f, "%.4f" % (fails[f] / float(n)), n, now) for f, n in sorted(totals.items())
                 if n >= MIN_EDITS]
        rows += [(ns, "coedit", "%s|%s" % pair, "", c, now) for pair, c in sorted(coedit.items()) if c >= MIN_SUPPORT]
        for typ, c in sorted(tiers.items()):
            tier, n = sorted(c.items(), key=lambda kv: (-kv[1], kv[0]))[0]
            if n >= MIN_SUPPORT:
                rows.append((ns, "tier", typ, tier, n, now))
        conn.execute("DELETE FROM patterns WHERE ns = ? AND kind IN (%s)" % ",".join("?" * len(KINDS)), (ns,) + KINDS)
        conn.executemany("INSERT INTO patterns (ns, kind, key, value, support, updated) VALUES (?, ?, ?, ?, ?, ?)", rows)
    return {"deduped": deduped, "edges": len(edges), "ranked": ranked, "patterns": len(rows)}
```

```python
#!/usr/bin/env python3
"""Memory hook stages for the dryas memory provider (owned core, sub-project B). Observe-only: read the hook
event on stdin, write the memory store, print nothing, always exit 0, stay inside a 2 s budget (busy timeout
0.5 s). A failure is one line in memory.jsonl.

  memhooks.py edit                       PostToolUse / PostToolUseFailure on Write|Edit|MultiEdit|NotebookEdit
  memhooks.py task                       SubagentStop: close the active task rows (tiers from dispatch.jsonl)
  memhooks.py session-start              SessionStart: session row; Claude memory files indexed as notes
  memhooks.py session-end [manual|auto]  SessionEnd (summary, consolidate) / PreCompact (summary only)
Sub-project D wires these into chain.json (mem-edit, mem-task, mem-session-start, mem-session-end)."""
import json
import os
import re
import sys
import time
from typing import Any, Dict, List, Optional, Tuple

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import jevlog

CONSOLIDATE_BEFORE_S = 1.0  # consolidate only if the stage has used less than this of its 2 s budget
NOTE_MAX_BYTES = 64 * 1024
STDIN_MAX = 1024 * 1024
TAIL_BYTES = 2 * 1024 * 1024
EDIT_TOOLS = ("Write", "Edit", "MultiEdit", "NotebookEdit")
STAGES = ("edit", "task", "session-start", "session-end")
_NAME = re.compile(r"^name:\s*(.+)$", re.M)


def _cwd(event: Dict[str, Any]) -> str:
    return event.get("cwd") or os.environ.get("CLAUDE_PROJECT_DIR") or os.getcwd()


def _find_up(start: str, rel: str) -> Optional[str]:
    d = os.path.abspath(start)
    while True:
        if os.path.exists(os.path.join(d, rel)):
            return d
        parent = os.path.dirname(d)
        if parent == d:
            return None
        d = parent


def repo_rel(path: str) -> str:
    """Path relative to the enclosing checkout (a .git dir, or a worktree's .git file), so worktrees of one repo
    agree on file names; the absolute path when there is none."""
    p = os.path.abspath(path)
    root = _find_up(os.path.dirname(p), ".git")
    return os.path.relpath(p, root).replace(os.sep, "/") if root else p.replace(os.sep, "/")


def _task_ids(cwd: str) -> List[str]:
    """Every active task id. Accepts both taskctx shapes: {"task", "co_tasks"?} and {"tasks"}."""
    ctx = taskctx.context(cwd)
    ids = [str(ctx["task"])] if ctx.get("task") else []
    for key in ("co_tasks", "tasks"):
        extra = ctx.get(key)
        for t in extra if isinstance(extra, list) else []:
            if str(t) not in ids:
                ids.append(str(t))
    return ids


def on_edit(conn, event: Dict[str, Any]) -> bool:
    if event.get("tool_name") not in EDIT_TOOLS:
        return False
    ti = event.get("tool_input") if isinstance(event.get("tool_input"), dict) else {}
    f = ti.get("file_path") or ti.get("notebook_path")
    if not isinstance(f, str) or not f:
        return False
    cwd = _cwd(event)
    resp = event.get("tool_response")
    failed = event.get("hook_event_name") == "PostToolUseFailure" or (
        isinstance(resp, dict) and (resp.get("success") is False or bool(resp.get("error"))))
    ids = _task_ids(cwd)
    with memdb.tx(conn):
        conn.execute("INSERT INTO edits (ns, session, task_id, file, ok, ts) VALUES (?, ?, ?, ?, ?, ?)",
                     (projns.project_ns(cwd), event.get("session_id"), ids[0] if len(ids) == 1 else None,
                      repo_rel(os.path.join(cwd, f)), 0 if failed else 1, time.time()))
    return True


def _dispatches(task_id: str) -> List[Dict[str, Any]]:
    try:
        with open(os.path.join(jevlog.log_dir(), "dispatch.jsonl"), "rb") as fh:
            fh.seek(0, 2)
            fh.seek(max(0, fh.tell() - TAIL_BYTES))
            data = fh.read()
    except OSError:
        return []
    out = []
    for line in data.decode("utf-8", "replace").splitlines():
        try:
            r = json.loads(line)
        except ValueError:
            continue
        if isinstance(r, dict) and str(r.get("task")) == task_id:
            out.append(r)
    return out


def _plan_section(cwd: str, task_id: str) -> Tuple[str, List[str]]:
    root = _find_up(cwd, os.path.join(".orchestrate", "PLAN.md"))
    if not root:
        return "", []
    try:
        with open(os.path.join(root, ".orchestrate", "PLAN.md"), encoding="utf-8") as fh:
            sec = planfile.sections(fh.read()).get(task_id.rsplit(":", 1)[-1])
    except (OSError, UnicodeDecodeError):
        return "", []
    if not sec:
        return "", []
    return sec.splitlines()[0].split(":", 1)[-1].strip()[:200], planfile.scope(sec)


def on_task(conn, event: Dict[str, Any]) -> int:
    cwd = _cwd(event)
    ns, now, n = projns.project_ns(cwd), time.time(), 0
    for tid in _task_ids(cwd):
        d = _dispatches(tid)
        tiers = [str(r["tier"]) for r in d if r.get("tier")]
        title, scope = _plan_section(cwd, tid)
        with memdb.tx(conn):
            conn.execute(
                "INSERT INTO tasks (task_id, ns, title, scope, tier_start, tier_end, outcome, started, ended, meta) "
                "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT(task_id) DO UPDATE SET "
                "title = CASE WHEN excluded.title != '' THEN excluded.title ELSE tasks.title END, "
                "scope = CASE WHEN excluded.scope != '[]' THEN excluded.scope ELSE tasks.scope END, "
                "tier_start = COALESCE(tasks.tier_start, excluded.tier_start), "
                "tier_end = COALESCE(excluded.tier_end, tasks.tier_end), "
                "outcome = excluded.outcome, ended = excluded.ended, meta = excluded.meta",
                (tid, ns, title, json.dumps(scope), tiers[0] if tiers else None, tiers[-1] if tiers else None,
                 "returned", now, now, json.dumps({"dispatches": len(d)})))
        n += 1
    return n


def notes_dir(cwd: str) -> Optional[str]:
    env = os.environ.get("DRYAS_NOTES_DIR")
    if env:
        return env if os.path.isdir(env) else None
    for root in (projns.main_root(cwd), cwd):
        if root:
            d = os.path.join(paths.projects_dir(), re.sub(r"[^A-Za-z0-9]", "-", os.path.abspath(root)), "memory")
            if os.path.isdir(d):
                return d
    return None


def _title(text: str, fallback: str) -> str:
    front = text.split("---", 2)[1] if text.startswith("---") and text.count("---") >= 2 else ""
    m = _NAME.search(front)
    return (m.group(1).strip() if m else fallback)[:200]


def index_notes(conn, ns: str, d: str) -> Dict[str, int]:
    counts = {"indexed": 0, "unchanged": 0, "skipped": 0, "removed": 0}
    existing: Dict[str, Tuple[int, Any]] = {}
    for r in conn.execute("SELECT id, key, meta FROM memory WHERE ns = ? AND source = 'claude-memory'", (ns,)):
        try:
            existing[r["key"]] = (r["id"], json.loads(r["meta"] or "{}").get("mtime"))
        except ValueError:
            existing[r["key"]] = (r["id"], None)
    seen = set()
    for name in sorted(os.listdir(d)):
        if not name.endswith(".md"):
            continue
        p = os.path.join(d, name)
        seen.add(p)
        try:
            st = os.stat(p)
            if st.st_size > NOTE_MAX_BYTES:
                counts["skipped"] += 1
                continue
            if p in existing and existing[p][1] == st.st_mtime:
                counts["unchanged"] += 1
                continue
            with open(p, encoding="utf-8") as fh:
                text = fh.read()
        except (OSError, UnicodeDecodeError):
            counts["skipped"] += 1
            continue
        title, body = _title(text, name[:-3]), redact.redact(text)
        meta, h = json.dumps({"mtime": st.st_mtime, "path": p}), memdb.content_hash(title, body)
        with memdb.tx(conn):
            if p in existing:
                conn.execute("UPDATE memory SET title = ?, body = ?, hash = ?, meta = ? WHERE id = ?",
                             (title, body, h, meta, existing[p][0]))
            else:
                conn.execute("INSERT INTO memory (ns, kind, title, body, tags, source, key, hash, meta, created) "
                             "VALUES (?, 'note', ?, ?, '[\"claude-memory\"]', 'claude-memory', ?, ?, ?, ?)",
                             (ns, title, body, p, h, meta, st.st_mtime))
        counts["indexed"] += 1
    gone = [v[0] for k, v in existing.items() if k not in seen]
    if gone:
        with memdb.tx(conn):
            conn.execute("DELETE FROM memory WHERE id IN (%s)" % ",".join("?" * len(gone)), gone)
        counts["removed"] = len(gone)
    return counts


def on_session_start(conn, event: Dict[str, Any]) -> Dict[str, int]:
    cwd = _cwd(event)
    ns = projns.project_ns(cwd)
    with memdb.tx(conn):
        conn.execute("INSERT OR IGNORE INTO sessions (id, ns, started) VALUES (?, ?, ?)",
                     (str(event.get("session_id") or "unknown"), ns, time.time()))
    d = notes_dir(cwd)
    return index_notes(conn, ns, d) if d else {"indexed": 0, "unchanged": 0, "skipped": 0, "removed": 0}


def summary(conn, ns: str, sid: str) -> str:
    e = conn.execute("SELECT COUNT(*), SUM(ok = 0), COUNT(DISTINCT file) FROM edits WHERE ns = ? AND session = ?",
                     (ns, sid)).fetchone()
    since = conn.execute("SELECT started FROM sessions WHERE id = ?", (sid,)).fetchone()[0] or 0
    t = conn.execute("SELECT COUNT(*) FROM tasks WHERE ns = ? AND ended >= ?", (ns, since)).fetchone()[0]
    m = conn.execute("SELECT COUNT(*) FROM memory WHERE ns = ? AND source = 'dryas' AND created >= ?",
                     (ns, since)).fetchone()[0]
    return "%d edits (%d failed) in %d files; %d tasks closed; %d memories stored" % (e[0], e[1] or 0, e[2], t, m)


def on_session_end(conn, event: Dict[str, Any], mode: Optional[str], t0: float) -> Dict[str, Any]:
    cwd = _cwd(event)
    ns, sid, now = projns.project_ns(cwd), str(event.get("session_id") or "unknown"), time.time()
    with memdb.tx(conn):
        conn.execute("INSERT OR IGNORE INTO sessions (id, ns, started) VALUES (?, ?, ?)", (sid, ns, now))
    text = summary(conn, ns, sid)
    with memdb.tx(conn):
        if mode is None:
            conn.execute("UPDATE sessions SET summary = ?, ended = ? WHERE id = ?", (text, now, sid))
        else:
            conn.execute("UPDATE sessions SET summary = ? WHERE id = ?", (text, sid))
    out: Dict[str, Any] = {"summary": text}
    if mode is None and time.monotonic() - t0 < CONSOLIDATE_BEFORE_S:
        out["consolidated"] = memlearn.consolidate(conn, ns)
    return out


def run(stage: str, event: Dict[str, Any], rest: List[str], t0: float) -> Any:
    if stage not in STAGES:
        raise ValueError("unknown stage %r" % stage[:32])
    conn = memdb.connect(busy_ms=HOOK_BUSY_MS)
    try:
        if stage == "edit":
            return on_edit(conn, event)
        if stage == "task":
            return on_task(conn, event)
        if stage == "session-start":
            return on_session_start(conn, event)
        return on_session_end(conn, event, rest[0] if rest and rest[0] in ("manual", "auto") else None, t0)
    finally:
        conn.close()


def main(argv: Optional[List[str]] = None) -> int:
    t0 = time.monotonic()
    argv = sys.argv[1:] if argv is None else argv
    stage = argv[0] if argv else ""
    try:
        event = json.loads(sys.stdin.buffer.read(STDIN_MAX).decode("utf-8", "replace") or "{}")
        if not isinstance(event, dict):
            raise ValueError("hook event is not an object")
        run(stage, event, argv[1:], t0)
    except Exception as e:
        jevlog.append("memory", dict(stage=stage[:32], error=type(e).__name__, detail=str(e)[:200]))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Import Ruflo memory into the Dryas store (owned core, sub-project B).

The Ruflo database is opened read-only and never changed. Every row is imported with source 'ruflo', its
namespace, timestamps and hit count, and the embedding kept as a blob for a later semantic provider. The
commands, feedback and session namespaces (hook noise) become kind 'archive', left out of default recall.
Idempotent: a row is keyed by its Ruflo id, and a row forgotten in Dryas is not brought back. Bodies pass
through redact.py (MEM-4); the report counts the redactions."""
import json
import os
import sqlite3
import sys
import time
import urllib.request
from collections import Counter
from typing import Any, Dict, Tuple

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import memdb
import paths
import redact

ARCHIVE_NS = ("commands", "feedback", "session")
MAX_BODY = 64 * 1024
BATCH = 1000
COLUMNS = ("id", "key", "namespace", "content", "type", "embedding", "tags", "metadata", "created_at",
           "updated_at", "last_accessed_at", "access_count", "status")


def default_source() -> str:
    return os.path.join(paths.ruflo_dir(), "memory", "agentdb-memory.db")


def _secs(v: Any) -> Any:
    """Ruflo writes milliseconds; older rows may hold seconds."""
    try:
        f = float(v)
    except (TypeError, ValueError):
        return None
    return f / 1000.0 if f > 1e11 else f


def _tags(v: Any) -> str:
    try:
        parsed = json.loads(v) if isinstance(v, str) else None
    except ValueError:
        parsed = None
    return json.dumps([str(t)[:64] for t in parsed][:20] if isinstance(parsed, list) else [])


def _ns(r: Dict[str, Any]) -> str:
    return str(r["namespace"]) if r["namespace"] else "default"


def _row(r: Dict[str, Any]) -> Tuple[Tuple[Any, ...], bool]:
    ns = _ns(r)
    content = str(r["content"] or "")[:MAX_BODY]
    body = redact.redact(content)
    title = str(r["key"] or "")[:200]
    emb = r["embedding"]
    emb = emb.encode("utf-8") if isinstance(emb, str) else emb
    meta = json.dumps(dict(ruflo_id=r["id"], type=r["type"], status=r["status"], updated=_secs(r["updated_at"]),
                           metadata=r["metadata"]))
    created = _secs(r["created_at"]) or time.time()
    try:
        hits = int(r["access_count"] or 0)
    except (TypeError, ValueError):
        hits = 0
    return ((ns, "archive" if ns in ARCHIVE_NS else "outcome", title, body, _tags(r["tags"]), "ruflo", str(r["id"]),
             memdb.content_hash(title, body), emb, meta, created, _secs(r["last_accessed_at"]), hits), body != content)


def import_ruflo(conn, src: str) -> Dict[str, Any]:
    if not os.path.isfile(src):
        raise FileNotFoundError(src)
    s = sqlite3.connect("file:%s?mode=ro" % urllib.request.pathname2url(os.path.abspath(src)), uri=True)
    try:
        cols = [c[1] for c in s.execute("PRAGMA table_info(memory_entries)")]
        if "id" not in cols:
            raise ValueError("no Ruflo memory_entries table in %s" % src)
        sel = ", ".join(c if c in cols else "NULL AS %s" % c for c in COLUMNS)
        before = conn.execute("SELECT COUNT(*) FROM memory WHERE source = 'ruflo'").fetchone()[0]
        src_counts: Counter = Counter()
        redacted = 0
        cur = s.execute("SELECT %s FROM memory_entries" % sel)
        while True:
            batch = cur.fetchmany(BATCH)
            if not batch:
                break
            rows = []
            for b in batch:
                r = dict(zip(COLUMNS, b))
                src_counts[_ns(r)] += 1
                vals, red = _row(r)
                rows.append(vals)
                redacted += red
            with memdb.tx(conn):
                gone = {(n, k) for n, k in conn.execute("SELECT ns, key FROM forgotten WHERE source = 'ruflo'")}
                conn.executemany(
                    "INSERT OR IGNORE INTO memory (ns, kind, title, body, tags, source, key, hash, embedding, meta, "
                    "created, accessed, hits) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
                    [v for v in rows if (v[0], v[6]) not in gone])
    finally:
        s.close()
    dst = {n: c for n, c in conn.execute("SELECT ns, COUNT(*) FROM memory WHERE source = 'ruflo' GROUP BY ns")}
    forgot = {n: c for n, c in conn.execute("SELECT ns, COUNT(*) FROM forgotten WHERE source = 'ruflo' GROUP BY ns")}
    nss = {n: {"source": c, "imported": dst.get(n, 0), "forgotten": forgot.get(n, 0)} for n, c in sorted(src_counts.items())}
    after = conn.execute("SELECT COUNT(*) FROM memory WHERE source = 'ruflo'").fetchone()[0]
    return {"namespaces": nss, "new": after - before, "redacted": redacted,
            "ok": all(v["imported"] + v["forgotten"] >= v["source"] for v in nss.values())}
```

```python
#!/usr/bin/env python3
"""Dryas memory CLI (owned core, sub-project B). The `dryas memory ...` wrapper (sub-project D) runs this.

  memcli.py [--db PATH] import-ruflo [--from PATH]   import Ruflo memory read-only; exit 1 if a namespace's
                                                     count does not verify, 2 if there is no Ruflo memory
  memcli.py [--db PATH] stats [--ns NS]              rows per namespace and kind
--db (or DRYAS_DB) picks the store; the default is ~/.dryas/dryas.db."""
import argparse
import os
import sqlite3
import sys
from typing import List, Optional

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import memdb
import ruflo_import


def main(argv: Optional[List[str]] = None) -> int:
    ap = argparse.ArgumentParser(prog="dryas memory")
    ap.add_argument("--db")
    sub = ap.add_subparsers(dest="cmd", required=True)
    imp = sub.add_parser("import-ruflo", help="import Ruflo memory (read-only source)")
    imp.add_argument("--from", dest="src")
    st = sub.add_parser("stats", help="rows per namespace and kind")
    st.add_argument("--ns")
    a = ap.parse_args(argv)
    if a.cmd == "import-ruflo":
        src = a.src or ruflo_import.default_source()
        if not os.path.isfile(src):
            print("no Ruflo memory at %s; nothing to import" % src)
            return 2
        conn = memdb.connect(a.db)
        try:
            rep = ruflo_import.import_ruflo(conn, src)
        except (sqlite3.Error, ValueError) as e:
            print("import failed: %s" % e, file=sys.stderr)
            return 1
        finally:
            conn.close()
        for ns, c in rep["namespaces"].items():
            print("%-24s %7d source %7d imported %5d forgotten%s" % (
                ns, c["source"], c["imported"], c["forgotten"], " (archive)" if ns in ruflo_import.ARCHIVE_NS else ""))
        print("%d new rows, %d bodies redacted: %s" % (rep["new"], rep["redacted"],
                                                        "verified" if rep["ok"] else "COUNT MISMATCH"))
        return 0 if rep["ok"] else 1
    conn = memdb.connect(a.db)
    try:
        sql = "SELECT ns, kind, COUNT(*) FROM memory" + (" WHERE ns = ?" if a.ns else "") + " GROUP BY ns, kind ORDER BY ns, kind"
        for ns, kind, n in conn.execute(sql, (a.ns,) if a.ns else ()):
            print("%-24s %-8s %7d" % (ns, kind, n))
    finally:
        conn.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Memory tools on the Dryas MCP server (owned core, sub-project B): memory_search, memory_store, memory_list,
memory_forget and swarm_init, with Ruflo's argument names accepted (key/value, namespace). jev_mcp.py lists
TOOLS and routes NAMES here. MEM-5: when the store cannot be opened every tool answers with degraded: true
instead of failing, and swarm_init still returns an id so /orchestrate continues."""
import json
import os
import sqlite3
import sys
import time
import uuid
from typing import Any, Dict, Optional, Tuple

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import memdb
import memstore
import projns

_NS = {"type": "string", "description": "Main repo folder name; default: the current project's"}
TOOLS = [
    {"name": "memory_search",
     "description": ("Search Dryas memory in one namespace (default: this project). Returns the top results with a "
                     "reason each, plus learned patterns that mention a query word (file failure rates, files edited "
                     "together, the tier that finished a task type). Archived hook noise only with include_archive."),
     "inputSchema": {"type": "object", "properties": {
         "query": {"type": "string"}, "namespace": _NS, "limit": {"type": "integer", "minimum": 1, "maximum": 20},
         "include_archive": {"type": "boolean"}}, "required": ["query"]}},
    {"name": "memory_store",
     "description": ("Store one task outcome (what worked, what failed) in Dryas memory. Secrets, diffs and bodies "
                     "over 4000 characters are refused (MEM-4). Ruflo's key/value are accepted as title/body."),
     "inputSchema": {"type": "object", "properties": {
         "title": {"type": "string"}, "body": {"type": "string"}, "key": {"type": "string"}, "value": {},
         "namespace": _NS, "kind": {"type": "string", "enum": list(memstore.STORE_KINDS)},
         "tags": {"type": "array", "items": {"type": "string"}}, "task_id": {"type": "string"},
         "commit": {"type": "string"}}}},
    {"name": "memory_list",
     "description": "List Dryas memory in one namespace, newest first (bodies cut to 300 characters).",
     "inputSchema": {"type": "object", "properties": {
         "namespace": _NS, "kind": {"type": "string"}, "limit": {"type": "integer", "minimum": 1, "maximum": 100},
         "offset": {"type": "integer", "minimum": 0}, "include_archive": {"type": "boolean"}}}},
    {"name": "memory_forget",
     "description": "Delete one memory by id. Notes mirror the Claude memory files: edit or delete the file instead.",
     "inputSchema": {"type": "object", "properties": {"id": {"type": "integer", "minimum": 1}, "namespace": _NS},
                     "required": ["id"]}},
    {"name": "swarm_init",
     "description": ("Open a task record for an orchestrated run and return its swarm id. Kept for Ruflo "
                     "compatibility: the topology is recorded, and the dispatch log stays the ledger of record."),
     "inputSchema": {"type": "object", "properties": {
         "topology": {"type": "string"}, "task": {"type": "string"},
         "scope": {"type": "array", "items": {"type": "string"}}, "maxAgents": {"type": "integer"},
         "strategy": {"type": "string"}, "namespace": _NS}}},
]
NAMES = frozenset(t["name"] for t in TOOLS)


def _ns(args: Dict[str, Any]) -> str:
    v = args.get("namespace")
    return projns.sanitize(v.strip()) if isinstance(v, str) and v.strip() else projns.project_ns()


def _int(v: Any, default: Optional[int], lo: int, hi: int) -> Optional[int]:
    if isinstance(v, bool):
        return default
    if isinstance(v, str) and v.strip().isdigit():
        v = int(v.strip())
    if not isinstance(v, int):
        return default
    return max(lo, min(v, hi))


def _opt(v: Any) -> Optional[str]:
    return str(v)[:200] if isinstance(v, (str, int)) and not isinstance(v, bool) and str(v).strip() else None


def _session() -> str:
    return os.environ.get("JEV_SESSION_ID") or "pid-%d" % os.getpid()


def _topology(args: Dict[str, Any]) -> str:
    t = args.get("topology")
    return t.strip()[:32] if isinstance(t, str) and t.strip() else "hierarchical"


def _swarm_id() -> str:
    return "swarm-" + uuid.uuid4().hex[:12]


def _degraded(name: str, ns: str, args: Dict[str, Any], err: Exception) -> Dict[str, Any]:
    out: Dict[str, Any] = {"degraded": True, "namespace": ns, "cause": type(err).__name__}
    if name in ("memory_search", "memory_list"):
        out["results"] = []
    elif name == "memory_store":
        out["stored"] = False
    elif name == "memory_forget":
        out["forgotten"] = False
    elif name == "swarm_init":
        out.update(swarmId=_swarm_id(), topology=_topology(args))
    return out


def _store(conn, ns: str, args: Dict[str, Any]) -> Tuple[Any, bool]:
    body = args.get("body", args.get("value"))
    if isinstance(body, (dict, list)):
        body = json.dumps(body)
    if not isinstance(body, str) or not body.strip():
        return {"stored": False, "reason": "body (or value) is required"}, True
    title = args.get("title", args.get("key"))
    title = str(title) if isinstance(title, (str, int)) and not isinstance(title, bool) else body.strip().splitlines()[0][:80]
    tags = args.get("tags")
    try:
        r = memstore.store(conn, ns, title, body, kind=args.get("kind") if args.get("kind") in memstore.STORE_KINDS else "outcome",
                           tags=[t for t in tags if isinstance(t, str)] if isinstance(tags, list) else [],
                           task_id=_opt(args.get("task_id")), commit=_opt(args.get("commit")))
    except memstore.Rejected as e:
        return {"stored": False, "reason": str(e)}, True
    return dict(r, namespace=ns), False


def _call(conn, name: str, ns: str, args: Dict[str, Any]) -> Tuple[Any, bool]:
    if name == "memory_search":
        res = memstore.search(conn, ns, str(args.get("query") or ""), limit=_int(args.get("limit"), 3, 1, 20),
                              include_archive=args.get("include_archive") is True, session=_session())
        return dict(res, namespace=ns), False
    if name == "memory_store":
        return _store(conn, ns, args)
    if name == "memory_list":
        rows = memstore.list_entries(conn, ns, kind=_opt(args.get("kind")), limit=_int(args.get("limit"), 20, 1, 100),
                                     offset=_int(args.get("offset"), 0, 0, 10 ** 9),
                                     include_archive=args.get("include_archive") is True)
        return {"results": rows, "namespace": ns}, False
    if name == "memory_forget":
        mid = args.get("id")
        if isinstance(mid, str) and mid.strip().isdigit():
            mid = int(mid.strip())
        if isinstance(mid, bool) or not isinstance(mid, int) or mid < 1:
            return {"forgotten": False, "reason": "id must be a positive integer"}, True
        status = memstore.forget(conn, ns, mid)
        if status == "note":
            return {"forgotten": False, "reason": "this mirrors a Claude memory file; edit or delete the file and it "
                                                  "leaves the index at the next session start"}, True
        return {"forgotten": status == "forgotten", "status": status}, False
    if name == "swarm_init":
        sid, topo = _swarm_id(), _topology(args)
        scope = args.get("scope")
        scope = [s for s in scope if isinstance(s, str)][:64] if isinstance(scope, list) else []
        with memdb.tx(conn):
            conn.execute("INSERT INTO tasks (task_id, ns, title, scope, started, meta) VALUES (?, ?, ?, ?, ?, ?)",
                         (sid, ns, str(args.get("task") or "")[:200], json.dumps(scope), time.time(),
                          json.dumps({"topology": topo, "kind": "swarm"})))
        return {"swarmId": sid, "topology": topo, "namespace": ns}, False
    return {"error": "unknown memory tool"}, True


def call(name: str, args: Any) -> Tuple[Any, bool]:
    args = args if isinstance(args, dict) else {}
    ns = _ns(args)
    try:
        conn = memdb.connect()
    except (sqlite3.Error, OSError) as e:
        return _degraded(name, ns, args, e), False
    try:
        return _call(conn, name, ns, args)
    except sqlite3.Error as e:
        return _degraded(name, ns, args, e), False
    finally:
        conn.close()
```
