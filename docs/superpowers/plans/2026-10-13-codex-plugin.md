# Dryas as a Codex CLI Plugin — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the Codex edition as the Codex plugin `dryas@dryas-workflow` plus a thin setup so `install.sh --harness codex [--codex-dir DIR]` sets up any `CODEX_HOME` with trusted hooks, and today's Codex copy install migrates to it.

**Architecture:** `codex/` becomes a self-contained plugin root (`.codex-plugin/plugin.json`, `skills/`, `hooks/hooks.json`, `.mcp.json`, and real copies of `claude/jev` and `claude/pstack` written by `tools/sync_codex_plugin.py`, guarded by a CI drift test). `.agents/plugins/marketplace.json` points Codex at `./codex`. Rules come from `jev/rules_inject.py --harness codex` at SessionStart. Setup lives in `install/codex_plugin_setup.py`; hook trust goes through `codex app-server` via `install/codex_appserver.py`; trust removal edits `config.toml` text via `install/codex_toml.py`. The Codex copy path (`install_codex`, the `~/.claude` runtime copy, `codex/hooks.fragment.json`) is deleted.

**Tech Stack:** Python 3.9+ stdlib, `unittest` (run with pytest), POSIX sh / PowerShell entry scripts, Codex plugin manifests (JSON), Codex app-server JSON-RPC over stdio (JSONL).

**Spec:** `docs/superpowers/specs/2026-10-13-codex-plugin-design.md` (decisions [J1]–[J7]). **Research:** `codex-plugin-research.md` (Codex CLI 0.162.1).

## Global Constraints

- Python 3.9+ stdlib only; tests are `unittest` classes run by `python -m pytest -q claude/jev/tests install/tests claude/ruflo/tests`. No `enterContext` (3.11+): use `p = mock.patch.dict(...); p.start(); tc.addCleanup(p.stop)`.
- CI matrix: macOS, Ubuntu, Windows × Python 3.9, 3.12. No Codex CLI in CI; use the fake from `install/tests/fakecli.py` (`make_codex`, Task 3).
- Plugin name `dryas`, marketplace name `dryas-workflow`, id `dryas@dryas-workflow` (same in both harnesses; different config homes).
- One source of truth: `claude/jev` and `claude/pstack`. `codex/jev` and `codex/pstack` are generated by `tools/sync_codex_plugin.py`; never hand-edit them. Every task that touches `claude/jev` or `claude/pstack` runs `python tools/sync_codex_plugin.py` and commits both.
- State root: `$DRYAS_DATA_ROOT`, default `~/.dryas`. Existing `JEV_*` env overrides still win. `chain.local.json` shared with Claude.
- Codex hook command form (frozen; part of the trust hash): `"${DRYAS_PY:-python3}" -X utf8 "${PLUGIN_ROOT}/jev/<script>.py" [args] --harness codex`.
- MCP form: `"cwd": "."`, relative args, secrets only via `env_vars`. Setup never writes a key into `config.toml`.
- Setup's own `config.toml` entries live in one block between `# >>> dryas plugin >>>` and `# <<< dryas plugin <<<`; trust tables are written by Codex (app-server) and removed by exact recorded key.
- Hooks fail open, except scope lock and Jev gate (keep deny behavior). `chain.py` always exits 0.
- Setup never writes the config block, trust, `chain.local.json` or `accounts.json` when the marketplace add, plugin add or plugin listing fails.
- Rules injected for Codex stay under 8,000 bytes for every component combination.
- Portability: no `/bin/sh`, `/bin/bash`, `/usr/bin/env bash`, `shell=True`, literal `"/tmp/"` in shipped files (`install/tests/test_portability.py`).
- Claude Code behavior is unchanged (except: `plugin_setup.add_account/remove_account` gain a `key`, purge counts `codex_dirs`, and `run_cmd` runs Codex before Claude).
- Minimum Codex: 0.162.1 (`MIN_CODEX`).
- Commits: conventional (`feat(...)`, `fix(...)`, `test(...)`, `docs(...)`), one per task minimum, explicit paths in `git add`.

# install/tests/test_codex_plugin_sync.py
"""codex/jev and codex/pstack are generated from claude/ (Codex copies plugins and drops symlinks)."""
import shutil, sys, tempfile, unittest
from pathlib import Path
ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "tools"))
import sync_codex_plugin as scp


class SyncTest(unittest.TestCase):
    def test_repo_has_no_drift(self):
        self.assertEqual(scp.drift(ROOT), [], "run: python tools/sync_codex_plugin.py")

    def test_plan_skips_tests_and_caches(self):
        rels = list(scp.plan(ROOT))
        self.assertIn("codex/jev/chain.py", rels)
        self.assertIn("codex/jev/chain.json", rels)
        self.assertIn("codex/pstack/benchmark-checklist.md", rels)
        self.assertFalse([r for r in rels if "/tests/" in r or "__pycache__" in r or r.endswith(".pyc")])

    def _tree(self):
        t = Path(tempfile.mkdtemp())
        self.addCleanup(shutil.rmtree, str(t), True)
        for s, _ in scp.TREES:
            (t / s).mkdir(parents=True)
        (t / "claude/jev/a.py").write_bytes(b"x = 1\n")
        (t / "claude/jev/tests").mkdir()
        (t / "claude/jev/tests/test_a.py").write_bytes(b"")
        (t / "claude/pstack/p.md").write_bytes(b"p\n")
        return t

    def test_drift_kinds_and_sync(self):
        t = self._tree()
        self.assertEqual(scp.drift(t), ["missing codex/jev/a.py", "missing codex/pstack/p.md"])
        self.assertEqual(sorted(scp.sync(t)), ["codex/jev/a.py", "codex/pstack/p.md"])
        self.assertEqual(scp.drift(t), [])
        (t / "codex/jev/a.py").write_bytes(b"x = 2\n")
        (t / "codex/jev/old.py").write_bytes(b"")
        self.assertEqual(scp.drift(t), ["differs codex/jev/a.py", "extra codex/jev/old.py"])
        scp.sync(t)
        self.assertFalse((t / "codex/jev/old.py").exists())
        self.assertEqual(scp.drift(t), [])

    def test_crlf_is_not_drift(self):
        t = self._tree()
        scp.sync(t)
        (t / "codex/jev/a.py").write_bytes(b"x = 1\r\n")
        self.assertEqual(scp.drift(t), [])

    def test_check_exit_code(self):
        t = self._tree()
        self.assertEqual(scp.main(["--check", "--repo", str(t)]), 1)
        scp.main(["--repo", str(t)])
        self.assertEqual(scp.main(["--check", "--repo", str(t)]), 0)


if __name__ == "__main__":
    unittest.main()

```python
#!/usr/bin/env python3
"""Copy the runtime the Codex plugin needs from claude/ (the source of truth) into codex/.
Codex installs a plugin by copying it and drops symlinks, so these are real copies.
  python tools/sync_codex_plugin.py           write codex/jev and codex/pstack
  python tools/sync_codex_plugin.py --check   list drift, exit 1 if any (CI)
"""
import argparse
import shutil
import sys
from pathlib import Path
from typing import Dict, List, Optional

REPO = Path(__file__).resolve().parents[1]
TREES = (("claude/jev", "codex/jev"), ("claude/pstack", "codex/pstack"))
SKIP_DIRS = {"tests", "__pycache__", ".pytest_cache"}


def _files(base: Path):
    for f in sorted(base.rglob("*")):
        rel = f.relative_to(base)
        if f.is_dir() or any(p in SKIP_DIRS for p in rel.parts[:-1]) or f.suffix == ".pyc":
            continue
        yield rel, f


def plan(repo: Path) -> Dict[str, Path]:
    return {(Path(dst) / rel).as_posix(): f for src, dst in TREES for rel, f in _files(Path(repo) / src)}


def _same(a: Path, b: Path) -> bool:
    return a.read_bytes().replace(b"\r\n", b"\n") == b.read_bytes().replace(b"\r\n", b"\n")


def _existing(repo: Path) -> List[str]:
    return [(Path(dst) / rel).as_posix() for _, dst in TREES for rel, _f in _files(Path(repo) / dst)
            if (Path(repo) / dst).is_dir()]


def drift(repo: Path) -> List[str]:
    want, out = plan(repo), []
    for rel, src in want.items():
        dest = Path(repo) / rel
        if not dest.is_file():
            out.append("missing " + rel)
        elif not _same(src, dest):
            out.append("differs " + rel)
    out += ["extra " + rel for rel in _existing(repo) if rel not in want]
    return sorted(out)


def sync(repo: Path) -> List[str]:
    changed = []
    for line in drift(repo):
        kind, rel = line.split(" ", 1)
        dest = Path(repo) / rel
        if kind == "extra":
            dest.unlink()
        else:
            dest.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(str(plan(repo)[rel]), str(dest))
        changed.append(rel)
    return changed


def main(argv: Optional[List[str]] = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    ap.add_argument("--check", action="store_true")
    ap.add_argument("--repo", default=str(REPO))
    a = ap.parse_args(argv)
    repo = Path(a.repo)
    if a.check:
        bad = drift(repo)
        for line in bad:
            print(line)
        if bad:
            print("codex/ is out of date: run python tools/sync_codex_plugin.py and commit the result")
        return 1 if bad else 0
    for rel in sync(repo):
        print("synced " + rel)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

"""Codex app-server client for hook trust (JSON-RPC over stdio, one JSON object per line, no "jsonrpc" header).
Setup uses it exactly as the Codex TUI's /hooks review does: initialize, initialized, hooks/list, config/batchWrite."""
import json
import queue
import subprocess
import threading
import time
from typing import Any, Dict, List, Optional

import platform_util as pu

CLIENT = {"name": "dryas_setup", "title": "Dryas setup", "version": "1.0.0"}


class AppServerError(Exception):
    pass


def hook_entries(result: Any) -> List[dict]:
    out: List[dict] = []
    if isinstance(result, dict):
        if "key" in result and "trustStatus" in result:
            return [result]
        for v in result.values():
            out += hook_entries(v)
    elif isinstance(result, list):
        for v in result:
            out += hook_entries(v)
    return out


def untrusted(entries: List[dict], plugin_id: str) -> List[dict]:
    return [e for e in entries if e.get("pluginId") == plugin_id and e.get("trustStatus") != "trusted"]


def trust_params(entries: List[dict]) -> dict:
    value = {e["key"]: {"trusted_hash": e["currentHash"]} for e in entries}
    return {"edits": [{"keyPath": "hooks.state", "value": value, "mergeStrategy": "upsert"}], "reloadUserConfig": True}


class AppServer:
    def __init__(self, env: Dict[str, str], argv: Optional[List[str]] = None, timeout: float = 30.0):
        self.timeout, self.next_id, self.q = timeout, 0, queue.Queue()  # type: ignore[var-annotated]
        try:
            self.p = subprocess.Popen(pu.resolve_argv(argv or ["codex", "app-server"]), stdin=subprocess.PIPE,
                                      stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, env=env,
                                      universal_newlines=True, encoding="utf-8", errors="replace")
        except OSError as e:
            raise AppServerError("cannot start codex app-server: %s" % e)
        threading.Thread(target=self._read, daemon=True).start()

    def _read(self) -> None:
        try:
            for line in self.p.stdout:  # type: ignore[union-attr]
                try:
                    self.q.put(json.loads(line))
                except ValueError:
                    continue
        finally:
            self.q.put(None)

    def _send(self, msg: dict) -> None:
        try:
            self.p.stdin.write(json.dumps(msg) + "\n")  # type: ignore[union-attr]
            self.p.stdin.flush()  # type: ignore[union-attr]
        except (OSError, ValueError) as e:
            raise AppServerError("codex app-server closed: %s" % e)

    def notify(self, method: str, params: Optional[dict] = None) -> None:
        self._send({"method": method, "params": params or {}})

    def request(self, method: str, params: dict) -> dict:
        self.next_id += 1
        rid, end = self.next_id, time.monotonic() + self.timeout
        self._send({"id": rid, "method": method, "params": params})
        while True:
            left = end - time.monotonic()
            if left <= 0:
                raise AppServerError("%s: no answer within %ss" % (method, self.timeout))
            try:
                msg = self.q.get(timeout=left)
            except queue.Empty:
                continue
            if msg is None:
                raise AppServerError("%s: codex app-server exited" % method)
            if msg.get("id") == rid and "method" not in msg:  # server requests and notifications are ignored
                if "error" in msg:
                    raise AppServerError("%s: %s" % (method, (msg["error"] or {}).get("message", msg["error"])))
                return msg.get("result") or {}

    def close(self) -> None:
        try:
            self.p.stdin.close()  # type: ignore[union-attr]
            self.p.wait(timeout=5)
        except (OSError, subprocess.TimeoutExpired):
            self.p.kill()

    def __enter__(self) -> "AppServer":
        return self

    def __exit__(self, *exc) -> None:
        self.close()


def _session(env: Dict[str, str], timeout: float) -> AppServer:
    s = AppServer(env, timeout=timeout)
    try:
        s.request("initialize", {"clientInfo": CLIENT})
        s.notify("initialized")
    except AppServerError:
        s.close()
        raise
    return s


def list_hooks(env: Dict[str, str], cwd: str, timeout: float = 30.0) -> List[dict]:
    with _session(env, timeout) as s:
        return hook_entries(s.request("hooks/list", {"cwds": [cwd]}))


def write_trust(env: Dict[str, str], entries: List[dict], timeout: float = 30.0) -> None:
    if not entries:
        return
    with _session(env, timeout) as s:
        s.request("config/batchWrite", trust_params(entries))

    parts = str(e.get("key", "")).split(":")

def trust_hooks(cdx: Path, rec: dict, yes: bool) -> bool:

    try:
        entries = ca.list_hooks(env, str(Path.home()))

        print("warning: Codex lists no Dryas hooks in %s" % cdx)
        print(TRUST_HINT)
        return False
    todo = ca.untrusted(entries, ID)
    if todo:
        print("Codex runs plugin hooks only after you trust them. Dryas hooks in %s:" % cdx)
        for e in todo:

            print(TRUST_HINT)
            return False

```python
"""Remove Dryas hook-trust entries from a Codex config.toml as text (Python 3.9 has no TOML parser).
Only exact recorded keys are touched; anything not found is returned, never guessed."""
import re
from typing import List, Tuple

_INLINE = re.compile(r"""^\s*(?:"([^"]+)"|'([^']+)')\s*=""")


def remove_trust(text: str, keys: List[str]) -> Tuple[str, List[str]]:
    want, found, out = set(keys), set(), []
    skip, section = False, None
    for line in text.splitlines(keepends=True):
        s = line.strip()
        if s.startswith("["):
            section, skip = s, False
            for k in want:
                if s in ('[hooks.state."%s"]' % k, "[hooks.state.'%s']" % k):
                    skip = True
                    found.add(k)
            if skip:
                continue
        elif skip:
            continue
        elif section == "[hooks.state]":
            m = _INLINE.match(line)
            k = m and (m.group(1) or m.group(2))
            if k in want:
                found.add(k)
                continue
        out.append(line)
    new = re.sub(r"\n{3,}", "\n\n", "".join(out))
    return (new if found else text), [k for k in keys if k not in found]
```
